DECLARE @CategoryIds TABLE (CategoryId INT);
INSERT INTO @CategoryIds (CategoryId)
VALUES (9), (10), (11), (12), (13), (14), (15),

WITH OrderedPosts AS (
    SELECT 
        Id AS PostId,
        ROW_NUMBER() OVER (ORDER BY Id) AS RowNum
    FROM Posts
),
MappedPosts AS (
    SELECT 
        PostId,
        ((RowNum - 1) / 1000) AS GroupIndex
    FROM OrderedPosts
),
FinalMapping AS (
    SELECT 
        p.PostId,
        c.CategoryId AS PostTextCategoryId
    FROM MappedPosts p
    JOIN (
        SELECT 
            CategoryId,
            ROW_NUMBER() OVER (ORDER BY CategoryId) - 1 AS GroupIndex
        FROM @CategoryIds
    ) c ON p.GroupIndex = c.GroupIndex
)
INSERT INTO PostsWithTextCategories (PostId, PostTextCategoryId)
SELECT PostId, PostTextCategoryId
FROM FinalMapping;
